# ==================================================================================================
# CMake addition for backend/tests/test_Threading.cpp
# ==================================================================================================
# Owner of the build files: this is the only change this test file needs. It adds a *new* executable
# rather than a case in `test_backend`, because the concurrency tests in it deliberately run thousands
# of interleavings and take orders of magnitude longer than the existing device-free suite; folding them
# into `test_backend` would put that cost on every developer who runs the suite for the type tables.
#
# Paste this into the `if (IS_HOST_PLATFORM)` block at the end of backend/CMakeLists.txt,
# after the existing `add_test(NAME test_${TARGET} ...)` / `set_tests_properties(...)` pair. `${TARGET}`
# is `backend` (set at the top of that file) and `gtest` is the wrapper target the application root
# provides and cmake/CoralGraphicsDependencies.cmake resolves for a standalone configure, so nothing
# here needs a new dependency.

add_executable(test_threading
    tests/test_asserts.h
    tests/test_Threading.cpp
    # A test binary needs its own `main`; `test_backend`'s is already in that target's sources and a
    # second copy would be a duplicate symbol. This shares the same one, which is what makes the two
    # executables behave identically under `--gtest_filter` and `--gtest_output`.
    tests/test_main.cpp
)

target_link_libraries(test_threading PRIVATE ${TARGET} gtest)

# `test_Threading.cpp` includes `private/backend/gl/GlDevice.h` to reach `GLDevice`'s command-buffer
# container and the `GLTextureMeta` / `GLBufferMeta` types it drives through `HandlePool`. That header
# lives under `include/private`, which is PRIVATE_HDR_DIR. It also includes `ShaderCross.h`, which lives
# in SRC_DIR, so the transitive include needs SRC_DIR too.
#
# Both are PRIVATE include directories on the library target, so neither is exported to a consumer and
# the test target has to name them explicitly - the same reason `test_backend` names `${SRC_DIR}` for
# ShaderCross.h. Verified by compiling `test_Threading.cpp` with exactly these two paths on the include
# line and no others from this directory.
target_include_directories(test_threading PRIVATE ${PRIVATE_HDR_DIR} ${SRC_DIR})

set_target_properties(test_threading PROPERTIES FOLDER Tests)

if (MSVC)
  target_compile_options(test_threading PRIVATE /utf-8 /W3)
else ()
  target_compile_options(test_threading PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif ()

# No fixture directory: this suite reads no files, so it has no working-directory dependency. It is
# registered with CTest like the backend suite so that it is not invisible to a `ctest` run - the
# failure mode that left the graphics suites unregistered for a while.
add_test(NAME test_threading COMMAND test_threading)
